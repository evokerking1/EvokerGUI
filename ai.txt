Assistant: 
Got it — in that case, the provider should **not** just fill a row with one item. It should let you define a reusable **component** made of multiple slots/items/actions, then place that component in a header/footer area automatically.

A good structure would be:

- `MenuComponent` = reusable group of items/actions.
- `Provider` = applies a component to a menu.
- `HeaderProvider` = places component starting at the first row.
- `FooterProvider` = places component starting at the last row.
- `SimpleMenu` = can register providers/components and apply them before opening.

This lets you make **one navbar** and reuse it across every menu.

Example:

```java
MenuComponent navBar = MenuComponent.row()
        .item(0, backItem, player -> openPreviousMenu(player))
        .item(4, homeItem, player -> openHomeMenu(player))
        .item(8, closeItem, Player::closeInventory);

SimpleMenu menu = new SimpleMenu(
        "Settings",
        SimpleMenu.Rows.FIVE,
        new FooterProvider(navBar)
);
```

The footer provider would automatically place that navbar in the last row, no matter if the menu has 3, 4, 5, or 6 rows.

package dev.evokerking.evokerGUI.menu.components;

import dev.evokerking.evokerGUI.menu.SimpleMenu;
import org.bukkit.entity.Player;
import org.bukkit.inventory.ItemStack;

import java.util.HashMap;
import java.util.Map;
import java.util.function.Consumer;

public class MenuComponent {
    private final Map<Integer, ComponentItem> items = new HashMap<>();

    public static MenuComponent row() {
        return new MenuComponent();
    }

    public MenuComponent item(int relativeSlot, ItemStack item) {
        return item(relativeSlot, item, player -> {});
    }

    public MenuComponent item(int relativeSlot, ItemStack item, Consumer<Player> action) {
        if (relativeSlot < 0 || relativeSlot > 8) {
            throw new IllegalArgumentException("Relative slot must be between 0 and 8 for a row component.");
        }

        items.put(relativeSlot, new ComponentItem(item, action));
        return this;
    }

    public void apply(SimpleMenu menu, int startSlot) {
        for (Map.Entry<Integer, ComponentItem> entry : items.entrySet()) {
            int slot = startSlot + entry.getKey();
            ComponentItem componentItem = entry.getValue();

            menu.setItem(slot, componentItem.item(), componentItem.action());
        }
    }

    private record ComponentItem(ItemStack item, Consumer<Player> action) {
    }
}
